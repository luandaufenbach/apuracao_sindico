```jsx
<Tabs items={['Daily', 'Weekly', 'Monthly', 'All']} />
<Tabs variant="segmented" items={[{ value: 'open', label: 'Open', count: 12 }, { value: 'closed', label: 'Closed' }]} />
<Tabs variant="underline" items={['Overview', 'Activity', 'Settings']} />
```
